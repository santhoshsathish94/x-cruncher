cmake_minimum_required(VERSION 3.16)
project(xcruncher VERSION 1.0.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

find_package(OpenMP)

add_library(xcruncher SHARED src/xcruncher.c)
target_include_directories(xcruncher PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(WIN32)
    target_compile_definitions(xcruncher PRIVATE XCRUNCHER_BUILD_DLL)
    if(MINGW)
        target_link_options(xcruncher PRIVATE -static)
    endif()
endif()
if(NOT MSVC)
    target_compile_options(xcruncher PRIVATE -O3 -Wall -Wextra)
    target_link_libraries(xcruncher PRIVATE m)
endif()
if(OpenMP_C_FOUND)
    target_link_libraries(xcruncher PUBLIC OpenMP::OpenMP_C)
endif()

add_library(xcruncher_static STATIC src/xcruncher.c)
target_include_directories(xcruncher_static PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(NOT MSVC)
    target_compile_options(xcruncher_static PRIVATE -O3 -Wall -Wextra)
    target_link_libraries(xcruncher_static PRIVATE m)
endif()
if(OpenMP_C_FOUND)
    target_link_libraries(xcruncher_static PUBLIC OpenMP::OpenMP_C)
endif()

enable_testing()
add_executable(test_xcruncher tests/test_xcruncher.c)
target_link_libraries(test_xcruncher PRIVATE xcruncher_static)
if(MSVC)
    target_link_options(test_xcruncher PRIVATE /STACK:8388608)
else()
    target_compile_options(test_xcruncher PRIVATE -O3)
    target_link_libraries(test_xcruncher PRIVATE m)
endif()
add_test(NAME xcruncher_unit_tests COMMAND test_xcruncher)

add_executable(bench_xcruncher bench/bench_xcruncher.c)
target_link_libraries(bench_xcruncher PRIVATE xcruncher_static)
if(MSVC)
    target_link_options(bench_xcruncher PRIVATE /STACK:8388608)
else()
    target_compile_options(bench_xcruncher PRIVATE -O3)
    target_link_libraries(bench_xcruncher PRIVATE m ${CMAKE_DL_LIBS})
endif()
